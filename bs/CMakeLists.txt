cmake_minimum_required(VERSION 3.10)
project(battleship C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/bin)
add_compile_options(-Wall -Wextra)
add_executable(battleship src/main.c src/judge.c src/board.c src/player.c)
add_executable(unit_tests tests/unit_tests.c src/board.c src/player.c)
enable_testing()
add_test(NAME unit COMMAND unit_tests)
add_test(NAME scenarios COMMAND sh ${CMAKE_SOURCE_DIR}/tests/run_tests.sh)
